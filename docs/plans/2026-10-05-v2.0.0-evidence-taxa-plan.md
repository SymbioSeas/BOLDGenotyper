```python
def _tree_sisters(tree_path, species_taxon: Dict[str, str], ids: List[str]) -> Dict[str, tuple]:
    """
    haplotype_id -> (sister taxon or '', support) for each of ids that is a tip
    of the tree. Branches with support below TREE_MIN_SUPPORT, or without a
    support value, are collapsed: walking from the tip toward the root, the
    first clade with enough support that holds another species-level
    haplotype is the sister clade. Its taxon is named when those haplotypes
    share one; the root never counts.
    """
    if not tree_path or not Path(tree_path).exists():
        return {}
    from Bio import Phylo
    tree = Phylo.read(str(tree_path), "newick")
    parent, tips_below = {}, {}
    for clade in tree.find_clades(order="postorder"):
        names = {clade.name} if clade.is_terminal() and clade.name else set()
        for child in clade.clades:
            parent[id(child)] = clade
            names |= tips_below[id(child)]
        tips_below[id(clade)] = names

    def support(clade):
        if clade.confidence is not None:
            return float(clade.confidence)
        try:
            return float(clade.name)
        except (TypeError, ValueError):
            return np.nan

    out = {}
    wanted = set(ids)
    for tip in tree.get_terminals():
        if tip.name not in wanted:
            continue
        out[tip.name] = ("", np.nan)
        node = tip
        while id(node) in parent:
            node = parent[id(node)]
            if node is tree.root:
                break
            s = support(node)
            if not s >= TREE_MIN_SUPPORT - _TOL:          # collapsed; NaN fails too
                continue
            others = [t for t in tips_below[id(node)] if t != tip.name and t in species_taxon]
            if others:
                taxa = {species_taxon[t] for t in others}
                if len(taxa) == 1:
                    out[tip.name] = (taxa.pop(), s)
                break
    return out
```

```python
EVIDENCE_SOURCE = "evidence"
NOT_APPLIED_COLUMNS = ["haplotype_id", "suggested_sp", "suggestion_basis", "why_not_applied"]


def _str(value) -> str:
    return "" if value is None or (not isinstance(value, str) and pd.isna(value)) else str(value)


def apply_evidence(taxonomy: pd.DataFrame, evidence: pd.DataFrame,
                   apply: bool = True) -> Tuple[pd.DataFrame, pd.DataFrame]:
    """
    Evidence taxa (design addendum of 2026-10-05, section 1).

    taxonomy: the label rule's table after curation (haplotype_id, assigned_sp,
    assignment_level, assignment_notes, assignment_source, leading_species,
    n_species_labels). evidence: compute_evidence output for that table.
    A haplotype the label rule left at genus (not curated) takes its
    suggested_sp when that is its leading species label, in the genus the
    label rule assigned: assignment_level species, assignment_source
    'evidence', notes 'evidence: <basis>; labels: <label rule note>'.
    Returns (taxonomy, suggestions not applied with why_not_applied). With
    apply False no taxon changes and every suggestion is returned.
    """
    from .taxon_labels import genus_of

    tax = taxonomy.copy()
    if "assignment_source" not in tax.columns:
        tax["assignment_source"] = "labels"
    if "assignment_notes" not in tax.columns:
        tax["assignment_notes"] = ""
    tax["assignment_notes"] = tax["assignment_notes"].astype(object)
    ev = evidence.set_index("haplotype_id")
    not_applied = []
    for idx, row in tax.iterrows():
        h = row["haplotype_id"]
        if h not in ev.index or row["assignment_level"] == "species" or row["assignment_source"] != "labels":
            continue
        suggested, basis = _str(ev.at[h, "suggested_sp"]), _str(ev.at[h, "suggestion_basis"])
        if not suggested:
            continue
        leading = _str(row.get("leading_species"))
        n_species = pd.to_numeric(pd.Series([row.get("n_species_labels")]), errors="coerce").fillna(0).iloc[0]
        if not apply:
            why = "evidence taxa off"
        elif not leading:
            why = "leading label tied" if n_species > 0 else "no species-level label"
        elif suggested != leading:
            why = "not the leading label"
        elif genus_of(suggested) != _str(row["assigned_sp"]).split(" ")[0]:
            why = "another genus than the label rule's"
        else:
            previous = _str(row["assignment_notes"])
            tax.at[idx, "assigned_sp"] = suggested
            tax.at[idx, "assignment_level"] = "species"
            tax.at[idx, "assignment_source"] = EVIDENCE_SOURCE
            tax.at[idx, "assignment_notes"] = f"evidence: {basis}" + (f"; labels: {previous}" if previous else "")
            continue
        not_applied.append({"haplotype_id": h, "suggested_sp": suggested,
                            "suggestion_basis": basis, "why_not_applied": why})
    return tax, pd.DataFrame(not_applied, columns=NOT_APPLIED_COLUMNS)
```

```python
SUGGESTIONS_FILE = "taxon_curation_suggestions.csv"
SUGGESTION_COLUMNS = ["seq_md5", "sequence", "taxon", "reason", "curator", "date",
                      "haplotype_id", "assigned_sp", "why_not_applied"]


def suggestions_table(not_applied: pd.DataFrame, haplotypes: pd.DataFrame,
                      sequences: Dict[str, str]) -> pd.DataFrame:
    """
    Suggestions not applied, as a curation file (taxon_curation): delete the
    rows to reject and pass the file to --taxon-curation. haplotypes has
    haplotype_id, seq_md5, assigned_sp and labels; sequences maps haplotype_id
    to its core sequence. The last three columns are for reading;
    load_curation ignores them.
    """
    haps = haplotypes.set_index("haplotype_id")

    def cell(h, col):
        return _str(haps.at[h, col]) if h in haps.index and col in haps.columns else ""

    rows = []
    for r in not_applied.to_dict("records"):
        h, labels = r["haplotype_id"], cell(r["haplotype_id"], "labels")
        rows.append({
            "seq_md5": cell(h, "seq_md5"),
            "sequence": sequences.get(h, "").replace("-", "").upper(),
            "taxon": r["suggested_sp"],
            "reason": f"suggested by evidence: {r['suggestion_basis']}" + (f"; labels: {labels}" if labels else ""),
            "curator": "", "date": "",
            "haplotype_id": h, "assigned_sp": cell(h, "assigned_sp"),
            "why_not_applied": r["why_not_applied"],
        })
    return pd.DataFrame(rows, columns=SUGGESTION_COLUMNS)
```

```python
def _build_tree(cfg, organism, dirs, haplotype_fasta, haplotype_stats, prov, step_warnings):
    """
    Build and root the haplotype tree (pipeline step 5.3; taxon evidence needs it
    before any step reads a taxon). Returns (tree path or None, intermediate
    prefix, output prefix). Failures are recorded as warnings, never raised.
    """
    intermediate_prefix = dirs['intermediate_phylo'] / organism
    output_prefix = dirs['phylogenetic'] / organism
    if not cfg.phylogenetic.build_tree:
        logger.info("  Phylogenetic tree building disabled in config")
        return None, intermediate_prefix, output_prefix
    try:
        logger.info("5.3: Building phylogenetic tree from haplotypes...")
        if not utils.check_external_tool("mafft"):
            step_warnings.record("5 tree", "MAFFT not found; tree not built")
            return None, intermediate_prefix, output_prefix
        if not utils.check_external_tool("fasttree"):
            step_warnings.record("5 tree", "FastTree not found; tree not built")
            return None, intermediate_prefix, output_prefix

        # Choose tips explicitly (P0-7): every haplotype with at least
        # --tree-min-members members and the minimum length, except
        # suspect haplotypes. Exclusions are listed with their reasons.
        records = list(SeqIO.parse(haplotype_fasta, "fasta"))
        suspect = {}
        if 'is_suspect' in haplotype_stats.columns:
            flagged = haplotype_stats[haplotype_stats['is_suspect'].fillna(False).astype(bool)]
            reasons = flagged['suspect_reasons'] if 'suspect_reasons' in flagged.columns else None
            suspect = {
                hid: (reasons.loc[i] if reasons is not None and isinstance(reasons.loc[i], str)
                      and reasons.loc[i] else 'flagged')
                for i, hid in flagged['haplotype_id'].items()
            }
        kept, excluded = phylogenetics.select_tree_haplotypes(
            records, cfg.phylogenetic.min_cluster_size,
            cfg.phylogenetic.min_consensus_length, suspect)
        excluded_path = dirs['phylogenetic'] / f"{organism}_excluded_tips.txt"
        with open(excluded_path, 'w') as f:
            f.write("haplotype_id\treason\n")
            for hid, reason in excluded:
                f.write(f"{hid}\t{reason}\n")
        tree_input = dirs['intermediate_phylo'] / f"{organism}_tree_input.fasta"
        SeqIO.write(kept, tree_input, "fasta")
        logger.info(f"  Tree tips: {len(kept)} of {len(records)} haplotypes "
                    f"({len(excluded)} excluded; see {excluded_path.name}); "
                    f"--tree-min-members {cfg.phylogenetic.min_cluster_size}, "
                    f"minimum length {cfg.phylogenetic.min_consensus_length} bp")
        prov.applied.update(
            tree_min_members=cfg.phylogenetic.min_cluster_size,
            tree_min_length_bp=cfg.phylogenetic.min_consensus_length,
            n_tree_tips=len(kept),
            n_tree_excluded=len(excluded),
        )

        # Tips are already selected, so build_phylogeny's own filter keeps everything.
        tree = phylogenetics.build_phylogeny(
            consensus_fasta=str(tree_input),
            output_prefix=str(intermediate_prefix),
            threads=cfg.n_threads,
            min_consensus_length=0,
            min_cluster_size=1,
            outgroup_fasta=str(cfg.phylogenetic.outgroup_fasta) if cfg.phylogenetic.outgroup_fasta else None,
            trim_alignment=cfg.phylogenetic.trim_alignment,
            trim_method=cfg.phylogenetic.trim_method
        )
        intermediate_tree = f"{intermediate_prefix}_tree.nwk"
        tree_path = f"{output_prefix}_tree.nwk"
        if Path(intermediate_tree).exists():
            Path(intermediate_tree).rename(tree_path)
        if tree is None or not Path(tree_path).exists():
            step_warnings.record("5 tree", f"tree building finished but {tree_path} was not written")
            return None, intermediate_prefix, output_prefix

        # Root on the outgroup when one is given, else at the midpoint;
        # support values stay on their branches (phylogenetics.root_on_tips)
        phy = cfg.phylogenetic
        rooting, rooted = {"method": "none"}, None
        if phy.outgroup_label:
            rooting = {"method": "outgroup_label", "outgroup": phy.outgroup_label}
            rooted = phylogenetics.reroot_tree_by_label(
                tree_file=tree_path, outgroup_label=phy.outgroup_label, output_file=tree_path)
        elif phy.outgroup_taxon:
            rooting = {"method": "outgroup_taxon", "outgroup": phy.outgroup_taxon}
            rooted = phylogenetics.reroot_tree_by_taxon(
                tree_file=tree_path,
                taxonomy_csv=str(dirs['taxonomy'] / f"{organism}_haplotype_taxonomy.csv"),
                taxon_query=phy.outgroup_taxon, output_file=tree_path)
        elif phy.outgroup_fasta:
            rooting = {"method": "outgroup_fasta", "outgroup": str(phy.outgroup_fasta)}
            rooted = phylogenetics.reroot_tree_by_tips(
                tree_path, [r.id for r in SeqIO.parse(str(phy.outgroup_fasta), "fasta")],
                output_file=tree_path, description="outgroup FASTA")
        if rooting["method"] != "none" and rooted is None:
            step_warnings.record(
                "5 tree reroot",
                f"rooting on {rooting['method']} '{rooting['outgroup']}' failed"
                + ("; midpoint-rooted instead" if phy.midpoint_root else "; tree left unrooted"))
            rooting = {"method": "none"}
        if rooting["method"] == "none" and phy.midpoint_root:
            if phylogenetics.midpoint_root_tree_file(tree_path):
                rooting = {"method": "midpoint"}
        prov.applied["tree_rooting"] = rooting
        logger.info(f"  Built phylogenetic tree: {tree_path}")
        return tree_path, intermediate_prefix, output_prefix
    except Exception as e:
        step_warnings.record("5 tree", e)
        return None, intermediate_prefix, output_prefix
```
