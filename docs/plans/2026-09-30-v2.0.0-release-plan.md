# BOLDGenotyper v2.0.0 Release Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Fix the v1.0.0 correctness and reproducibility bugs (P0), the query-tool problems (Q1–Q10), the default interpretations (P1), redundancy (P2) and polish (P3), and release the result as 2.0.0.

**Architecture:** See `2026-09-30-v2.0.0-release-design.md`. Three new focused modules carry the cross-cutting logic: `provenance.py` (run provenance and step warnings), `haplotype_ids.py` (`seq_md5`, display IDs, FASTA headers) and `record_status.py` (one status per input record). A fourth, `taxon_labels.py`, classifies species labels for the species counts, the query tool and label-conflict evidence. `dereplication.py` keys haplotypes by `seq_md5` internally and assigns display IDs once, at the end.

**Tech Stack:** Python 3.9+ (tested on 3.13), pandas, Biopython, geopandas/cartopy, MAFFT, FastTree, pytest.

```python
"""Synthetic BOLD-portal-style TSV for end-to-end tests.

Sequences are built from codons whose first two bases are not "TA" or "AG".
Any base in the third position of such a codon gives a sense codon under the
vertebrate mitochondrial code (stops: TAA, TAG, AGA, AGG), so every planted
variant passes ORF validation. Variants differ only at third codon positions.
"""

import csv
import random
from pathlib import Path

_SAFE_CODONS = [
    a + b + c
    for a in "ACGT" for b in "ACGT" for c in "ACGT"
    if (a + b) not in ("TA", "AG")
]
_NEXT_BASE = {"A": "C", "C": "G", "G": "T", "T": "A"}

COLUMNS = [
    "processid", "sampleid", "bin_uri", "species", "genus", "family",
    "country/ocean", "coord", "coord_source", "marker_code", "nuc",
]


def base_sequence(n_codons=200, seed=7):
    """Return a reproducible stop-free sequence of n_codons codons."""
    rng = random.Random(seed)
    return "".join(rng.choice(_SAFE_CODONS) for _ in range(n_codons))


def mutate(seq, codon_indices):
    """Change the third base of each listed codon to the next base."""
    s = list(seq)
    for i in codon_indices:
        pos = 3 * i + 2
        s[pos] = _NEXT_BASE[s[pos]]
    return "".join(s)


A = base_sequence()
B = mutate(A, [10, 20])
C = mutate(A, range(40, 48))
D = mutate(A, range(60, 75))

# processid -> (sequence, species label)
_RECORDS = [(f"SYN{i:03d}", A, "Genus alpha") for i in range(1, 11)]
_RECORDS += [(f"SYN{i:03d}", A[3:], "Genus alpha") for i in (11, 12)]
_RECORDS += [(f"SYN{i:03d}", B, "Genus beta") for i in range(13, 17)]
_RECORDS += [("SYN017", C, "Genus alpha"), ("SYN018", C, "Genus alpha"), ("SYN019", C, "Genus")]
_RECORDS += [
    ("SYN020", D, "Genus gamma"),
    ("SYN021", mutate(A, [30]), "Genus alpha"),
    ("SYN022", mutate(A, [10]), "Genus beta"),
    ("SYN023", A[:480], "Genus alpha"),
    ("SYN024", A[:270] + "N" * 60 + A[330:], "Genus alpha"),
    ("SYN025", "", "Genus alpha"),
]

EXPECTED = {
    "n_input": len(_RECORDS) + 1,   # one duplicated row
    "n_unique": len(_RECORDS),
    "status": {
        **{f"SYN{i:03d}": "assigned" for i in range(1, 21)},
        "SYN021": "assigned_corrected",
        "SYN022": "excluded_error_singleton",
        "SYN023": "excluded_core_coverage",
        "SYN024": "failed_qc",
        "SYN025": "no_sequence",
    },
    "haplotype_sequences": {A, B, C, D},   # uppercase, ungapped cores
    "members": {"A": 13, "B": 4, "C": 3, "D": 1},   # A: 12 exact + 1 corrected
}

_COORDS = [
    ("[12.5, -45.25]", ""),
    ("[-3, 150]", "GPS"),
    ("", ""),
    ("[-20.1, 134.2]", "Coordinates from country centroid"),
    ("[0, 0]", ""),
]


def make_synthetic_bold_tsv(path):
    """Write the synthetic TSV to path and return it."""
    path = Path(path)
    rows = []
    for i, (pid, seq, label) in enumerate(_RECORDS):
        coord, source = _COORDS[i % len(_COORDS)]
        rows.append({
            "processid": pid,
            "sampleid": f"S-{pid}",
            "bin_uri": "BOLD:SYN0001",
            "species": label if " " in label else "",
            "genus": label.split()[0],
            "family": "Syntheticidae",
            "country/ocean": "Atlantis" if coord else "",
            "coord": coord,
            "coord_source": source,
            "marker_code": "COI-5P",
            "nuc": seq,
        })
    rows.append(dict(rows[0]))  # duplicate processid, as in real portal exports
    with open(path, "w", newline="", encoding="utf-8") as f:
        writer = csv.DictWriter(f, fieldnames=COLUMNS, delimiter="\t")
        writer.writeheader()
        writer.writerows(rows)
    return path
```

```python
# boldgenotyper/provenance.py
"""Run provenance and step warnings for a BOLDGenotyper run.

StepWarnings collects failures of optional steps and figures so that a run
reports them in one place (the end of the log and run_provenance.json) instead
of only in scattered log lines.
"""

import logging
from typing import Dict, List

logger = logging.getLogger(__name__)


class StepWarnings:
    """Collect step and figure failures during a pipeline run."""

    def __init__(self):
        self._items: List[Dict[str, str]] = []

    def record(self, step, message):
        """Record a failure of `step` and log it as a warning."""
        item = {"step": str(step), "message": str(message)}
        self._items.append(item)
        logger.warning(f"[{item['step']}] {item['message']}")

    def __len__(self):
        return len(self._items)

    def as_list(self):
        return [dict(i) for i in self._items]

    def log_summary(self, log, organism):
```

```python
# boldgenotyper/haplotype_ids.py
"""Stable haplotype keys and display IDs.

A haplotype's stable key is seq_md5: the md5 of its uppercase, ungapped core
sequence. It is identical across runs for an identical sequence. Display IDs
(H0001, H0002, ...) carry no counts; they are numbered by member count
(descending) and then seq_md5, so identical input gives identical IDs.
"""

import hashlib
import re
from typing import Dict, Optional

_HEADER_FIELD = re.compile(r"\b(seq_md5|n_members)=(\S+)")


def seq_md5(seq: str) -> str:
    """Return the md5 hex digest of seq, uppercased with '-' and '.' removed."""
    core = str(seq).upper().replace("-", "").replace(".", "")
    return hashlib.md5(core.encode("ascii")).hexdigest()


def assign_display_ids(counts: Dict[str, int]) -> Dict[str, str]:
    """Map seq_md5 -> display ID, ordered by count descending, then seq_md5."""
    order = sorted(counts, key=lambda k: (-counts[k], k))
    width = max(4, len(str(len(order))))
    return {key: f"H{i:0{width}d}" for i, key in enumerate(order, start=1)}


def format_fasta_header(haplotype_id: str, md5: str, n_members: int) -> str:
    """Return a FASTA header line (without '>') for a haplotype."""
    return f"{haplotype_id} seq_md5={md5} n_members={int(n_members)}"


def parse_fasta_header(description: str) -> Dict[str, Optional[object]]:
    """Parse a haplotype FASTA header; fields that are absent are None."""
    text = str(description or "").lstrip(">").strip()
    parts = text.split(None, 1)
    fields = dict(_HEADER_FIELD.findall(text))
    n = fields.get("n_members")
    return {
        "haplotype_id": parts[0] if parts else None,
        "seq_md5": fields.get("seq_md5"),
        "n_members": int(n) if n is not None and n.isdigit() else None,
    }
```

```python
def assign_error_singletons(removed, retained_records, min_singleton_distance, assign_corrected=True):
    """Assign each removed error singleton's members to the nearest retained haplotype.

    Returns a DataFrame with one row per member processid: processid, status,
    haplotype_key (None unless assigned), nearest_key, distance_to_nearest,
    tied_keys (';'-joined, empty unless tied).
    """
    rows = []
    retained = [(r.id, str(r.seq).upper()) for r in retained_records]
    for _key, seq, members in removed:
        seq = seq.upper()
        dists = [(round(_compute_distance(seq, rseq), 12), rkey) for rkey, rseq in retained]
        best = min(d for d, _ in dists) if dists else float("nan")
        nearest = [k for d, k in dists if d == best]
        tied = len(nearest) > 1
        status = "assigned_corrected" if (assign_corrected and not tied and nearest) else "excluded_error_singleton"
        for pid in members:
            rows.append({
                "processid": pid,
                "status": status,
                "haplotype_key": nearest[0] if status == "assigned_corrected" else None,
                "nearest_key": nearest[0] if len(nearest) == 1 else None,
                "distance_to_nearest": best,
                "tied_keys": ";".join(nearest) if tied else "",
            })
    return pd.DataFrame(rows, columns=["processid", "status", "haplotype_key", "nearest_key",
                                       "distance_to_nearest", "tied_keys"])
```

```python
# boldgenotyper/record_status.py
"""One assignment status per unique input record.

Statuses, in order of precedence:
  no_sequence              no sequence in the input
  failed_qc                failed pre-alignment QC (reason kept)
  excluded_core_coverage   ungapped core shorter than the per-sequence cutoff
  assigned                 core identical to a retained haplotype
  assigned_corrected       error-filtered singleton assigned to its nearest retained haplotype
  excluded_error_singleton error-filtered singleton equidistant from 2+ retained haplotypes,
                           or correction disabled
"""

from typing import Sequence, Set

import pandas as pd

STATUSES = (
    "assigned", "assigned_corrected", "excluded_error_singleton",
    "excluded_core_coverage", "failed_qc", "no_sequence",
)

COLUMNS = ["processid", "status", "haplotype_id", "nearest_haplotype",
           "distance_to_nearest", "status_note"]


def build_record_status(unique_ids: Sequence[str], sequence_ids: Set[str], qc: pd.DataFrame,
                        exact_mapping: pd.DataFrame, excluded_core: Sequence[str],
                        error_singletons: pd.DataFrame, singleton_distance: float) -> pd.DataFrame:
    """Return one row per unique record with its status and a true note."""
    qc_idx = qc.set_index("processid") if len(qc) else pd.DataFrame(columns=["qc_pass", "qc_fail_reason"])
    exact = dict(zip(exact_mapping["processid"], exact_mapping["haplotype_id"]))
    excluded = set(excluded_core)
    err = error_singletons.set_index("processid") if len(error_singletons) else None
    rows = []
    for pid in unique_ids:
        row = {"processid": pid, "status": None, "haplotype_id": None,
               "nearest_haplotype": None, "distance_to_nearest": float("nan"), "status_note": ""}
        if pid not in sequence_ids:
            row.update(status="no_sequence", status_note="No sequence in the input")
        elif pid in qc_idx.index and not bool(qc_idx.loc[pid, "qc_pass"]):
            reason = qc_idx.loc[pid, "qc_fail_reason"] or "failed QC"
            row.update(status="failed_qc", status_note=f"Failed QC: {reason}")
        elif pid in excluded:
            row.update(status="excluded_core_coverage",
                       status_note="Ungapped core shorter than the per-sequence coverage cutoff")
        elif pid in exact:
            row.update(status="assigned", haplotype_id=exact[pid],
                       status_note="Core sequence identical to a retained haplotype")
        elif err is not None and pid in err.index:
            e = err.loc[pid]
            d = float(e["distance_to_nearest"])
            row["distance_to_nearest"] = d
            if e["status"] == "assigned_corrected":
                row.update(status="assigned_corrected", haplotype_id=e["haplotype_id"],
                           nearest_haplotype=e["nearest_haplotype"],
                           status_note=(f"Singleton within {singleton_distance} of another haplotype; "
                                        f"assigned to nearest retained haplotype {e['haplotype_id']} "
                                        f"(distance {d:.4f})"))
            elif e["tied_haplotypes"]:
                tied = ", ".join(str(e["tied_haplotypes"]).split(";"))
                row.update(status="excluded_error_singleton",
                           status_note=f"Error singleton equidistant ({d:.4f}) from {tied}")
            else:
                row.update(status="excluded_error_singleton", nearest_haplotype=e["nearest_haplotype"],
                           status_note=(f"Error singleton; correction disabled "
                                        f"(nearest {e['nearest_haplotype']}, distance {d:.4f})"))
        else:
            raise ValueError(f"No status could be determined for record {pid}")
        rows.append(row)
    return pd.DataFrame(rows, columns=COLUMNS)
```

```python
# boldgenotyper/taxon_labels.py
"""Classify species labels and count species under explicit definitions.

label_level():
  species  two or more words naming a species ("Genus alpha")
  open     open nomenclature ("Genus sp. 1", "Genus cf. alpha", "Genus aff. alpha")
  genus    a single word ("Genus")
  missing  empty or NaN
"""

import math
from typing import Dict, Iterable, Optional

import pandas as pd

_OPEN_TOKENS = {"sp", "sp.", "spp", "spp.", "cf", "cf.", "aff", "aff.", "nr", "nr.", "gr", "gr."}


def _clean(label) -> str:
    if label is None or (isinstance(label, float) and math.isnan(label)):
        return ""
    text = str(label).strip()
    return "" if text.lower() in ("nan", "none") else text


def label_level(label) -> str:
    """Return 'species', 'open', 'genus' or 'missing' for a species label."""
    parts = _clean(label).split()
    if not parts:
        return "missing"
    if len(parts) == 1:
        return "genus"
    if any(p.lower() in _OPEN_TOKENS for p in parts[1:]):
        return "open"
    return "species"


def genus_of(label) -> Optional[str]:
    """Return the first word of a label, or None for a missing label."""
    parts = _clean(label).split()
    return parts[0] if parts else None


def species_count_summary(input_labels: Iterable, qc_passed_labels: Iterable,
                          haplotype_taxonomy: pd.DataFrame,
                          majority_threshold: float) -> Dict[str, int]:
    """Species counts under the definitions documented in the output README."""
    inputs = {_clean(x) for x in input_labels} - {""}
    qc = {_clean(x) for x in qc_passed_labels} - {""}
    tax = haplotype_taxonomy.copy()
    tax["assigned_sp"] = tax["assigned_sp"].map(_clean)
    species_level = tax[tax["assigned_sp"].map(label_level) == "species"]
    confident = species_level[species_level["majority_fraction"] >= majority_threshold]
    return {
        "n_species_labels_input": len(inputs),
        "n_species_labels_species_level": sum(1 for x in inputs if label_level(x) == "species"),
        "n_species_labels_qc_passed": len(qc),
        "n_species_haplotype_assigned": species_level["assigned_sp"].nunique(),
        "n_species_confident": confident["assigned_sp"].nunique(),
    }
```

```python
_VERSION_PATTERNS = {
    "mafft": (["mafft", "--version"], r"v?(\d+\.\d+\S*)"),
    "fasttree": (["FastTree"], r"FastTree(?: version)? (\d+\.\d+\S*)"),
    "trimal": (["trimal", "--version"], r"trimAl v?(\S+)"),

def file_md5(path) -> str:
    h = hashlib.md5()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()


def count_data_rows(tsv_path) -> int:
    """Number of non-empty lines after the header."""
    with open(tsv_path, "r", encoding="utf-8", errors="replace") as f:
        return max(sum(1 for line in f if line.strip()) - 1, 0)


def tool_versions():
    out = {}
    for name, (cmd, pattern) in _VERSION_PATTERNS.items():
        exe = shutil.which(cmd[0]) or shutil.which(name)
        if exe is None:
            out[name] = None
            continue
        try:
            res = subprocess.run([exe] + cmd[1:], capture_output=True, text=True,
                                 timeout=20, stdin=subprocess.DEVNULL)
            m = re.search(pattern, (res.stdout or "") + (res.stderr or ""), flags=re.IGNORECASE)
            out[name] = m.group(1) if m else "unknown"
        except (OSError, subprocess.SubprocessError):
            out[name] = "unknown"
    return out


def git_commit_info(package_dir) -> dict:
    """Commit and dirty flag when the package is imported from a git checkout."""
    repo = Path(package_dir).resolve().parent
    if not (repo / ".git").exists() or shutil.which("git") is None:
        return {}
    try:
        sha = subprocess.run(["git", "--no-optional-locks", "-C", str(repo), "rev-parse", "HEAD"],
                             capture_output=True, text=True, timeout=10).stdout.strip()
        dirty = subprocess.run(["git", "--no-optional-locks", "-C", str(repo), "status", "--porcelain",
                                "--untracked-files=no"], capture_output=True, text=True, timeout=10).stdout.strip()
    except (OSError, subprocess.SubprocessError):
        return {}
    return {"git_commit": sha, "git_dirty": bool(dirty)} if sha else {}


def _now():
    return datetime.now().astimezone().isoformat(timespec="seconds")


class RunProvenance:
    """Everything needed to reproduce and audit one pipeline run."""

    def __init__(self, cfg, argv, input_path, output_dir, download_date=None):
        from . import __version__
        from .config import _convert_paths_to_strings
        self.version = __version__
        full = _convert_paths_to_strings(cfg.to_dict())
        self.config = {k: v for k, v in full.items() if k not in NOT_APPLIED}
        self.argv = list(argv)
        self.input_path = Path(input_path)
        self.output_dir = Path(output_dir)
        self.download_date = download_date
        self.started_at = _now()
        self.finished_at = None
        self.status = "running"
        self.applied = {}
        self.data_notes = {}
        self.warnings = StepWarnings()
        self.input = {
            "path": str(self.input_path),
            "md5": file_md5(self.input_path),
            "n_rows": count_data_rows(self.input_path),
            "download_date": download_date,
        }
        self._tools = tool_versions()
        self._git = git_commit_info(Path(__file__).parent)

    def to_dict(self):
        return {
            "schema_version": 1,
            "boldgenotyper_version": self.version,
            **self._git,
            "python_version": platform.python_version(),
            "platform": platform.platform(),
            "argv": self.argv,
            "command_line": shlex.join(self.argv),
            "input": dict(self.input),
            "output_dir": str(self.output_dir),
            "config": self.config,
            "applied": self.applied,
            "not_applied": NOT_APPLIED,
            "tools": self._tools,
            "seeds": {"fasttree": FASTTREE_SEED},
            "started_at": self.started_at,
            "finished_at": self.finished_at,
            "status": self.status,
            "warnings": self.warnings.as_list(),
            "data_notes": self.data_notes,
        }

    def finish(self, success=True):
        self.finished_at = _now()
        if not success:
            self.status = "failed"
        else:
            self.status = "completed_with_warnings" if len(self.warnings) else "completed"

    def write(self, path):
        path = Path(path)
        path.write_text(json.dumps(self.to_dict(), indent=2, default=str))
        return path
```

```python
_PREVIOUS_RUN_FILES = ("run_provenance.json", "records.csv", "haplotypes.fasta")
_PREVIOUS_RUN_GLOBS = ("*_pipeline_parameters.json", "*_annotated.csv")


def find_previous_run(output_dir):
    """Names of files or folders in output_dir that show an earlier run.

    The current run's own log is not a marker, and neither is an empty folder
    (a run that failed early can leave empty folders behind).
    """
    out = Path(output_dir)
    if not out.is_dir():
        return []
    found = [name for name in _PREVIOUS_RUN_FILES if (out / name).exists()]
    for pattern in _PREVIOUS_RUN_GLOBS:
        found.extend(p.name for p in out.glob(pattern))
    hap = out / "haplotypes"
    if hap.is_dir() and any(hap.iterdir()):
        found.append("haplotypes/")
    return sorted(set(found))
```

```python
def select_tree_haplotypes(records, min_members, min_length, suspect):
    """Choose tree tips; return (kept_records, [(haplotype_id, reason), ...]).
```

```python
_COORD_PAIR = re.compile(r"^\s*\[?\s*(-?\d+(?:\.\d+)?)\s*,\s*(-?\d+(?:\.\d+)?)\s*\]?\s*$")


def parse_coord_column(df, coord_col="coord", lat_col="lat", lon_col="lon"):
    """Add float lat/lon for every row with a parseable coordinate pair.

    Accepts "[lat, lon]" (current BOLD portal exports; integers or decimals)
    and "lat, lon" (older exports). Existing non-empty lat/lon columns win.
    """
    out = df.copy()
    lat = pd.Series(np.nan, index=out.index, dtype=float)
    lon = pd.Series(np.nan, index=out.index, dtype=float)
    if coord_col in out.columns:
        parsed = out[coord_col].fillna("").astype(str).str.extract(_COORD_PAIR)
        lat = pd.to_numeric(parsed[0], errors="coerce")
        lon = pd.to_numeric(parsed[1], errors="coerce")
    for col, values in ((lat_col, lat), (lon_col, lon)):
        if col in out.columns:
            existing = pd.to_numeric(out[col], errors="coerce")
            out[col] = existing.where(existing.notna(), values)
        else:
            out[col] = values
    return out
```
